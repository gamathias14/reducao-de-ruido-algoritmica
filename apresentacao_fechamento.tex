\begin{frame}{O objetivo deixou de ser só comparar algoritmos}
    \kicker{Problema}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.49\textwidth}
            \textbf{Pergunta inicial}
            \begin{itemize}
                \item Qual método reduz ruído sem destruir inteligibilidade?
                \item Qual custo é compatível com processamento causal?
                \item STFT e Wavelet são suficientes?
            \end{itemize}
        \end{column}
        \begin{column}{0.49\textwidth}
            \textbf{Pergunta de integração}
            \begin{itemize}
                \item O áudio tratado pode virar um microfone Windows?
                \item O transporte preserva blocos e sequência?
                \item O endpoint sustenta continuidade temporal?
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.35cm}
    \begin{block}{Tese do fechamento}
        Qualidade do DSP, integração funcional e continuidade temporal são
        evidências diferentes e precisam de gates separados.
    \end{block}
\end{frame}

\begin{frame}{A arquitetura separa DSP, transporte e endpoint}
    \kicker{Arquitetura}
    \centering
    \begin{tikzpicture}[
        node distance=0.13cm,
        bloco/.style={draw=deepblue, rounded corners=2pt, fill=softgray,
            text width=1.95cm, minimum height=1.18cm, align=center,
            font=\scriptsize},
        seta/.style={-{Latex[length=2mm]}, thick, uspblue}
    ]
        \node[bloco] (fonte) {Fonte ou\\replay};
        \node[bloco, right=of fonte] (dsp) {DSP causal\\RNNoise};
        \node[bloco, right=of dsp] (pcm) {PCM16\\16 kHz, 20 ms};
        \node[bloco, right=of pcm] (ctl) {Controle\\PCM v1};
        \node[bloco, right=of ctl] (drv) {SYSVAD\\ring buffer};
        \node[bloco, right=of drv] (ep) {Endpoint\\cliente Windows};
        \draw[seta] (fonte) -- (dsp);
        \draw[seta] (dsp) -- (pcm);
        \draw[seta] (pcm) -- (ctl);
        \draw[seta] (ctl) -- (drv);
        \draw[seta] (drv) -- (ep);
    \end{tikzpicture}

    \vspace{0.55cm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Contrato do bloco}
            \begin{itemize}
                \item 320 amostras, 640 bytes;
                \item sequência monotônica;
                \item IOCTLs e produtor exclusivo;
                \item \textit{underrun} vira silêncio, não repetição.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Observabilidade}
            \begin{itemize}
                \item recepção e processamento;
                \item fila local e fila do driver;
                \item escrita, descartes e \textit{underruns};
                \item captura WASAPI e scheduler.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{As camadas isoladas funcionam com folga mensurável}
    \kicker{Evidência}
    \centering
    \metric{1,951 ms}{p99 do RNNoise no host em 30.000 blocos}
    \hfill
    \metric{2,021 ms}{p99 do self-test RNNoise na VM em 3.000 blocos}
    \hfill
    \metric{1.000}{blocos por perna no transporte determinístico}

    \vspace{0.55cm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.49\textwidth}
            \textbf{Processamento e transporte}
            \begin{itemize}
                \item zero bloco acima de 20 ms nos gates isolados;
                \item zero erro de sequência, CRC, framing ou perda;
                \item zero erro de escrita nos gates aceitos.
            \end{itemize}
        \end{column}
        \begin{column}{0.49\textwidth}
            \textbf{Microfone virtual funcional}
            \begin{itemize}
                \item tom conhecido de 440 Hz;
                \item WAV de 12 s, mono, PCM16, 16 kHz;
                \item captura por cliente externo;
                \item silêncio após parada do produtor.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{A continuidade temporal falhou depois do DSP}
    \kicker{Limite da VM}
    \begin{columns}[T,onlytextwidth]

            \begin{tikzpicture}[
                ok/.style={draw=softgreen, fill=softgreen!8, rounded corners=2pt,

                fail/.style={draw=alertred, fill=alertred!8, rounded corners=2pt,
                    minimum width=5.8cm, minimum height=0.72cm, align=center}
            ]
                \node[ok] at (0,2.4) {Entrada host-cadenciada íntegra};
                \node[ok] at (0,1.4) {RNNoise dentro do orçamento};
                \node[ok] at (0,0.4) {IOCTL e fila do driver sem erro};
                \node[fail] at (0,-0.6) {Consumo WaveRT e captura com pausas};
            \end{tikzpicture}
        \end{column}
        \begin{column}{0.42\textwidth}
            \textbf{Hipóteses testadas}
            \begin{itemize}
                \item polling, timer, \textit{yield} e spin;
                \item 2, 3 e 4 vCPUs;
                \item afinidade e prioridade;
                \item antecipação do envio;
                \item WASAPI orientado a evento.
            \end{itemize}
            \vspace{0.15cm}
            \textcolor{alertred}{\textbf{Nenhuma eliminou perdas e
            \textit{underruns} de forma reproduzível.}}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{A demonstração final precisa de três provas separadas}
    \kicker{Demonstração}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.31\textwidth}
            \begin{block}{1. DSP audível}
                A/B entre fala ruidosa e fala tratada pelo RNNoise, antes da
                ponte.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{2. Endpoint funcional}
                Sinal sintético conhecido alimentando o SYSVAD e capturado por
                cliente Windows.
            \end{block}
        \end{column}
        \begin{column}{0.31\textwidth}
            \begin{block}{3. Integração técnica}
                Hashes, contadores e logs de processamento, transporte e
                escrita no driver.
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.42cm}
    \centering
    \colorbox{alertred!10}{%
        \parbox{0.91\textwidth}{\centering
        \textcolor{alertred}{\textbf{Não usar uma captura longa da VM como
        prova de tempo real ponta a ponta e não editar áudio para mascarar
        lacunas.}}}}
\end{frame}

\begin{frame}{A contribuição é técnica e também metodológica}
    \kicker{Fechamento}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \textbf{Contribuições}
            \begin{itemize}
                \item benchmark multicritério reproduzível;
                \item decisão perceptual coerente com os gates;
                \item RNNoise persistente por blocos;
                \item PCM v1 e microfone virtual funcional;
                \item instrumentação que localiza a fronteira da falha.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Limitações}
            \begin{itemize}
                \item escuta privada com amostra limitada;
                \item ambiente virtualizado não preservou tempo;
                \item driver não certificado como correto;
                \item latência física não medida;
                \item protótipo não é produto distribuível.
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.25cm}
    \begin{block}{Resultado final}
        Protótipo acadêmico integrado, funcional em suas camadas isoladas e
        limitado temporalmente no ambiente virtualizado.
    \end{block}
\end{frame}

\begin{frame}{A validação seguinte deve ocorrer em Windows nativo}
    \kicker{Próximo passo}
    \begin{enumerate}
        \item bancada secundária, isolada, com imagem restaurável;
        \item baseline HDA ou microfone físico;
        \item instalação controlada do SYSVAD em modo de teste;
        \item tom determinístico e captura do endpoint;
        \item matriz pareada bypass/RNNoise;
        \item gate de zero perda e zero \textit{underrun};
        \item medição física de latência;
        \item somente então fala controlada e escuta cega.
    \end{enumerate}

    \vspace{0.3cm}
    \centering
    {\color{alertred}\bfseries Um driver de teste pode causar tela azul: a
    bancada deve ser recuperável e não pode ser o computador principal.}
\end{frame}
